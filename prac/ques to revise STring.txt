public class Solution {
    public ListNode detectCycle(ListNode head) {
        ListNode s1=head;
        ListNode f1=head;
        boolean isCycle=false;
        while(f1!=null && f1.next!=null){
            s1=s1.next;
            f1=f1.next.next;
            if(s1==f1){
                isCycle=true;
                break;
            }
        }
        if(!isCycle){
            return null;
        }
        s1=head;
        ListNode prev=null;
        while(s1!=f1){
            s1=s1.next;
            prev=f1;// prev store prev old value before it move to next
            f1=f1.next;
        }
        prev.next=null; // remove cycle
        return s1;
        
    }
}

int arr[]={3,2,3};
        int n=arr.length;
        int candidate=0;
        int count=0;
        for(int num:arr){
            if(count==0){
                candidate=num;
            }
            if(num==candidate){
                count++;
            }else{
                count--;
            }
        }
        count=0;
        for(int num:arr){
            if(num==candidate){
              count++;
            }
        }
        if(count>n/2){
            return candidate;
        }
        return -1;
    }
